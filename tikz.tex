\usetikzlibrary{graphs,graphdrawing,positioning,quotes,shapes.geometric}
\usegdlibrary{layered}
\tikz[nodes=black, decision/.style={diamond, draw=black, thick, fill=yellow!50},terminator/.style={rectangle,draw=black, rounded corners=1.5ex, thick, fill=yellow!50},process/.style={rectangle, draw=black, thick, fill=yellow!50}]
\graph[layered layout, grow'=right, edges=black]{

start[terminator] -- desc -> check[decision],
go[process] -> check[decision],
check[decision] -> end[terminator],
};
